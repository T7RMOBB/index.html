const express = require('express'); const bcrypt = require('bcryptjs'); const jwt = require('jsonwebtoken'); const app = express(); app.use(express.json()); const JWT_SECRET = 'sua_chave_secreta_super_segura'; // --- BANCO DE DADOS EM MEMÓRIA --- const bancoDados = { usuarios: [], emprestimos: [] }; // Admin Padrão (async () => { const senhaHash = await bcrypt.hash('admin123', 10); bancoDados.usuarios.push({ id: 1, nome: 'Administrador', email: 'admin@sistema.com', senha_hash: senhaHash, role: 'admin' }); console.log('✅ Admin pronto: admin@sistema.com | Senha: admin123'); })(); // --- MIDDLEWARE DE AUTENTICAÇÃO --- function autenticarToken(req, res, next) { const authHeader = req.headers['authorization']; const token = authHeader && authHeader.split(' ')[1]; if (!token) return res.status(401).json({ error: 'Acesso negado' }); jwt.verify(token, JWT_SECRET, (err, user) => { if (err) return res.status(403).json({ error: 'Token inválido' }); req.user = user; next(); }); } // --- ROTAS DA API --- // 1. Login app.post('/api/login', async (req, res) => { const { email, senha } = req.body; const usuario = bancoDados.usuarios.find(u => u.email === email); if (!usuario) return res.status(400).json({ error: 'Usuário não encontrado' }); const senhaValida = await bcrypt.compare(senha, usuario.senha_hash); if (!senhaValida) return res.status(400).json({ error: 'Senha incorreta' }); const token = jwt.sign( { id: usuario.id, nome: usuario.nome, role: usuario.role }, JWT_SECRET, { expiresIn: '8h' } ); res.json({ token, role: usuario.role, nome: usuario.nome, id: usuario.id }); }); // 2. Criar Cliente e Empréstimo app.post('/api/admin/criar-emprestimo', autenticarToken, async (req, res) => { if (req.user.role !== 'admin') { return res.status(403).json({ error: 'Acesso restrito ao Admin' }); } const { nome, email, senha, telefone, endereco, valor, taxaJuros, dias, dataInicio } = req.body; // Criar ou localizar o login do cliente let usuario = bancoDados.usuarios.find(u => u.email === email); if (!usuario) { const senhaHash = await bcrypt.hash(senha, 10); usuario = { id: bancoDados.usuarios.length + 1, nome, email, senha_hash: senhaHash, role: 'cliente', telefone, endereco }; bancoDados.usuarios.push(usuario); } // Cálculos Financeiros const valorNum = parseFloat(valor); const taxaNum = parseFloat(taxaJuros); const diasNum = parseInt(dias); const valorJuros = valorNum * (taxaNum / 100); const valorTotal = valorNum + valorJuros; const valorDiaria = valorTotal / diasNum; // Cálculo da Data Final const inicio = new Date(dataInicio); const dataFinal = new Date(inicio); dataFinal.setDate(inicio.getDate() + diasNum); const novoEmprestimo = { id: bancoDados.emprestimos.length + 1, clienteId: usuario.id, valorEmpresado: valorNum, taxaJuros: taxaNum, valorTotal, valorDiaria, diasTotal: diasNum, dataInicio, dataFinal: dataFinal.toISOString().split('T')[0], status: 'Ativo' }; bancoDados.emprestimos.push(novoEmprestimo); res.status(201).json({ message: 'Empréstimo cadastrado com sucesso!', emprestimo: novoEmprestimo }); }); // 3. Listar todos os Empréstimos (Admin) app.get('/api/admin/emprestimos', autenticarToken, (req, res) => { if (req.user.role !== 'admin') return res.status(403).json({ error: 'Acesso negado' }); const lista = bancoDados.emprestimos.map(e => { const cliente = bancoDados.usuarios.find(u => u.id === e.clienteId); return { ...e, clienteNome: cliente ? cliente.nome : 'N/A', clienteTelefone: cliente ? cliente.telefone : 'N/A', clienteEndereco: cliente ? cliente.endereco : 'N/A' }; }); res.json(lista); }); // 4. Ver Empréstimos do Cliente Logado app.get('/api/cliente/meus-emprestimos', autenticarToken, (req, res) => { const meusEmprestimos = bancoDados.emprestimos.filter(e => e.clienteId === req.user.id); res.json(meusEmprestimos); }); // --- INTERFACE VISUAL (HTML) --- app.get('/', (req, res) => { res.send(` Sistema de Gestão de Crédito

`); }); app.listen(3000, () => { console.log('🚀 Servidor rodando em http://localhost:3000'); });
